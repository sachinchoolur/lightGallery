<Fragment slot="react">

```tsx
// `iframe: true` on the item replaces the vanilla selector wiring.
const slides = [
    {
        src: 'https://www.lightgalleryjs.com/',
        iframe: true,
        iframeTitle: 'lightGallery',
    },
    { src: 'https://maps.google.com/maps?q=…&output=embed', iframe: true },
    { src: 'files/sample.pdf', iframe: true },
];

<LightGallery slides={slides} open={open} onClose={() => setOpen(false)} />;
```
