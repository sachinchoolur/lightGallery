<Fragment slot="react">

```tsx
// One gallery, mixed slide kinds: images, video URLs and iframes are
// item fields rather than markup.
const slides = [
    { src: 'img/1.jpg', thumb: 'img/1-t.jpg' },
    { src: 'https://youtu.be/IUN664s7N-c', poster: 'img/yt-poster.jpg' },
    { src: 'https://example.com', iframe: true, thumb: 'img/site-t.jpg' },
];

<LightGallery slides={slides} plugins={[Thumbnail, Video]} />;
```
